export default function AddStudioForm() {

  const [values, setValues] = useState<FormValues>(INITIAL_VALUES);

  const [imageFile, setImageFile] = useState<File | null>(null);
  const [imagePreview, setImagePreview] = useState("");
  const [saving, setSaving] = useState(false);

  const updateValue = (field: keyof FormValues, value: string) => {
    setValues((current) => ({ ...current, [field]: value }));

  const toggleAmenity = (amenity: string) => {
    setAmenities((current) =>
      current.includes(amenity)
        ? current.filter((item) => item !== amenity)
        : [...current, amenity]
    );
  };

  const addCustomAmenity = () => {
    const value = customAmenity.trim();

    if (!value) return;

    if (amenities.some((item) => item.toLowerCase() === value.toLowerCase())) {
      toast.error("This amenity has already been added.");
      return;

    if (!file) return;

    if (!["image/jpeg", "image/png", "image/webp"].includes(file.type)) {
      toast.error("Choose a JPG, PNG, or WebP image.");
      return;
    }

      toast.error("Image size must be 5 MB or less.");
      return;

    setImageFile(file);

    const reader = new FileReader();

    reader.onload = () => {
      setImagePreview(String(reader.result ?? ""));

    reader.readAsDataURL(file);

      fileInputRef.current.value = "";

    if (values.title.trim().length < 3) {
      toast.error("Studio name must contain at least 3 characters.");
      return false;

      return false;

import type { Studio } from "@/lib/studios";

const API_URL = process.env.NEXT_PUBLIC_API_URL?.replace(/\/+$/, "");

type StudioApiResponse = {
  success?: boolean;
  data?: Record<string, unknown> | Record<string, unknown>[];
  message?: string;
};

export type StudioQuery = {
  search?: string;
  category?: string;
  location?: string;
  maxPrice?: number;
  sortBy?: string;
  page?: number;
  limit?: number;
};

export type CreateStudioPayload = {
  title: string;
  category: string;
  location: string;
  hourlyRate: number;
  securityDeposit: number;
  capacity: number;
  image: string;
  amenities: string[];
  overview: string;
};

function normalizeStudio(raw: Record<string, unknown>): Studio {
  return {
    id: String(raw._id ?? raw.id ?? ""),
    title: String(raw.title ?? ""),
    category: String(raw.category ?? ""),
    location: String(raw.location ?? ""),
    hourlyRate: Number(raw.hourlyRate ?? 0),
    securityDeposit: Number(raw.securityDeposit ?? 0),
    rating: Number(raw.rating ?? 0),
    reviewCount: Number(raw.reviewCount ?? 0),
    capacity: Number(raw.capacity ?? 0),
    image: String(raw.image ?? ""),
    amenities: Array.isArray(raw.amenities)
      ? raw.amenities.map(String)
      : [],
    overview: String(raw.overview ?? ""),
    featured: Boolean(raw.featured),
    ownerName: String(raw.ownerName ?? ""),
    ownerEmail: String(raw.ownerEmail ?? ""),
    ownerPhotoURL: String(raw.ownerPhotoURL ?? ""),
    status: String(raw.status ?? "Available"),
  };
}

async function parseResponse(response: Response): Promise<StudioApiResponse> {
  const contentType = response.headers.get("content-type") || "";

  if (!contentType.includes("application/json")) {
    const responseText = await response.text();

    console.error("Studio API returned a non-JSON response:", {
      status: response.status,
      contentType,
      preview: responseText.slice(0, 200),
    });

    throw new Error(
      `The Studio API returned an unexpected response (${response.status}). Check your API URL and backend route.`
    );
  }

  return (await response.json()) as StudioApiResponse;
}

async function requestStudios(
  endpoint: string,
  options?: RequestInit
): Promise<Studio[]> {
  if (!API_URL) {
    throw new Error(
      "NEXT_PUBLIC_API_URL is missing. Check your frontend .env.local file."
    );
  }

  const response = await fetch(`${API_URL}/api/studios${endpoint}`, {
    ...options,
    headers: {
      ...(options?.body instanceof FormData
        ? {}
        : { "Content-Type": "application/json" }),
      ...options?.headers,
    },
  });

  const result = await parseResponse(response);

  if (!response.ok || result.success === false) {
    throw new Error(result.message || "Unable to load studios.");
  }

  if (Array.isArray(result.data)) {
    return result.data.map(normalizeStudio);
  }

  if (result.data && typeof result.data === "object") {
    return [normalizeStudio(result.data)];
  }

  return [];
}

export async function getStudios(
  params: StudioQuery = {}
): Promise<Studio[]> {
  const searchParams = new URLSearchParams();

  Object.entries(params).forEach(([key, value]) => {
    if (value !== undefined && value !== null && value !== "") {
      searchParams.set(key, String(value));
    }
  });

  const query = searchParams.toString();

  return requestStudios(query ? `?${query}` : "");
}

export async function getFeaturedStudios(): Promise<Studio[]> {
  return requestStudios("/featured");
}

export async function getStudioById(id: string): Promise<Studio | null> {
  if (!id) return null;

  try {
    const studios = await requestStudios(`/${encodeURIComponent(id)}`);
    return studios[0] ?? null;
  } catch (error) {
    console.error("Unable to load studio details:", error);
    return null;
  }
}

export async function createStudio(
  payload: CreateStudioPayload
): Promise<Studio> {
  if (!API_URL) {
    throw new Error(
      "NEXT_PUBLIC_API_URL is missing. Check your frontend .env.local file."
    );
  }

  const response = await fetch(`${API_URL}/api/studios`, {
    method: "POST",
    credentials: "include",
    headers: {
      "Content-Type": "application/json",
    },
    body: JSON.stringify(payload),
  });

  const result = await parseResponse(response);

  if (!response.ok || result.success === false) {
    throw new Error(result.message || "Unable to create studio.");
  }

  if (!result.data || Array.isArray(result.data)) {
    throw new Error("The server did not return the created studio.");
  }

  return normalizeStudio(result.data);
}

const API_URL = (
  process.env.NEXT_PUBLIC_API_URL || "http://localhost:5000"
).replace(/\/+$/, "");

export type AuthUser = {
  id: string;
  name: string;
  email: string;
  photoURL?: string;
  role?: string;
  createdAt?: string;
};

export type LoginData = {
  email: string;
  password: string;
};

export type RegisterData = {
  name: string;
  email: string;
  photoURL: string;
  password: string;
};

export type ProfileUpdateData = {
  name?: string;
  photoURL?: string;
};

type AuthResponse = {
  success: boolean;
  message: string;
  user: AuthUser;
};

async function parseResponse<T>(response: Response): Promise<T> {
  const result = await response.json().catch(() => ({}));

  if (!response.ok) {
    throw new Error(
      result.message || "Something went wrong. Please try again."
    );
  }

  return result as T;
}

async function authRequest(
  endpoint: string,
  data?: object
): Promise<AuthResponse> {
  let response: Response;

  try {
    response = await fetch(`${API_URL}/api/auth/${endpoint}`, {
      method: "POST",
      headers: data
        ? { "Content-Type": "application/json" }
        : undefined,
      credentials: "include",
      cache: "no-store",
      body: data ? JSON.stringify(data) : undefined,
    });
  } catch {
    throw new Error(
      "Cannot connect to StudioSpace server. Make sure the backend is running on port 5000."
    );
  }

  return parseResponse<AuthResponse>(response);

export async function uploadProfileImage(file: File): Promise<string> {
  if (!["image/jpeg", "image/png", "image/webp"].includes(file.type)) {
    throw new Error("Please select a JPG, PNG, or WebP image.");
  }

  if (file.size > 5 * 1024 * 1024) {
    throw new Error("Your image must be smaller than 5 MB.");
  }

  const formData = new FormData();
  formData.append("image", file);

  let response: Response;

  try {
    response = await fetch(`${API_URL}/api/uploads/profile-image`, {
      method: "POST",
      credentials: "include",
      cache: "no-store",
      body: formData,
    });
  } catch {
    throw new Error("Unable to connect to the image upload service.");
  }

  const result = await parseResponse<{ imageURL: string }>(response);

  if (!result.imageURL) {
    throw new Error("The image service did not return an image URL.");
  }

  return result.imageURL;
}

export async function getCurrentUser(): Promise<AuthUser | null> {
  try {
    const response = await fetch(`${API_URL}/api/auth/me`, {
      method: "GET",
      credentials: "include",
      cache: "no-store",
    });

    if (!response.ok) return null;

    const result = await response.json();
    return result.user ?? null;
  } catch {
    return null;
  }
}

"use client";

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useState,
  type ReactNode,
} from "react";

import {
  getCurrentUser,
  logoutUser,
  type AuthUser,
} from "@/services/auth";

type AuthContextType = {
  user: AuthUser | null;
  loading: boolean;
  setUser: (user: AuthUser | null) => void;
  refreshUser: () => Promise<void>;
  logout: () => Promise<void>;
};

const AuthContext = createContext<AuthContextType | undefined>(undefined);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<AuthUser | null>(null);
  const [loading, setLoading] = useState(true);

  const refreshUser = useCallback(async () => {
    const currentUser = await getCurrentUser();
    setUser(currentUser);
  }, []);

  useEffect(() => {
    let active = true;

    async function loadUser() {
      try {
        const currentUser = await getCurrentUser();

        if (active) {
          setUser(currentUser);
        }
      } finally {
        if (active) {
          setLoading(false);
        }
      }
    }

    loadUser();

    return () => {
      active = false;
    };
  }, []);

  const logout = useCallback(async () => {
    await logoutUser();
    setUser(null);
  }, []);

  return (
    <AuthContext.Provider
      value={{
        user,
        loading,
        setUser,
        refreshUser,
        logout,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() {
  const context = useContext(AuthContext);

  if (!context) {
    throw new Error("useAuth must be used inside AuthProvider.");
  }

  return context;
}
